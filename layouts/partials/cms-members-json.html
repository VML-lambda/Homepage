{{- /* CMS 편집 화면용 연구실 멤버 목록을 /admin/members.json 으로 생성합니다. - Papers / Standard Contributions 미리보기 상단에 "저자 이름 참고용" 목록으로 표시 (static/admin/preview.js) - 홈 템플릿(layouts/index.html)에서 한 번 호출되며, 빌드할 때마다 현재 멤버로 다시 생성됩니다. - 순서는 홈페이지와 동일 (교수 → Ph.D. → M.S. → Intern → Alumni, 각 그룹은 weight 순) 출력 형식: { "groups": [ { "label": "Ph.D. Students", "members": [ { "name": "...", "korName": "..." } ] } ] } */ -}} {{- $groups := slice -}} {{- with site.GetPage "/members/professor" -}} {{- $groups = $groups | append (dict "label" "Professor" "members" (slice (dict "name" .Title "korName" (.Params.korName | default "")))) -}} {{- end -}} {{- with site.GetPage "/members/students" -}} {{- $students := .Pages.ByParam "weight" -}} {{- range $deg := slice (slice "PhD" "Ph.D. Students") (slice "MS" "M.S. Students") (slice "Intern" "Interns") -}} {{- $list := slice -}} {{- range where $students "Params.degree" (index $deg 0) -}} {{- $list = $list | append (dict "name" .Title "korName" (.Params.korName | default "")) -}} {{- end -}} {{- with $list }}{{ $groups = $groups | append (dict "label" (index $deg 1) "members" .) }}{{ end -}} {{- end -}} {{- end -}} {{- with site.GetPage "/members/alumni" -}} {{- $list := slice -}} {{- range .Pages.ByParam "weight" -}} {{- $list = $list | append (dict "name" .Title "korName" (.Params.korName | default "")) -}} {{- end -}} {{- with $list }}{{ $groups = $groups | append (dict "label" "Alumni" "members" .) }}{{ end -}} {{- end -}} {{- $res := resources.FromString "admin/members.json" (dict "groups" $groups | jsonify) -}} {{- $res.Publish -}}
